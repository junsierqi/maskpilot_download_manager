cmake_minimum_required(VERSION 3.25)
project(MaskPilot VERSION 0.3.0 LANGUAGES CXX)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 NEW)
endif()
option(MASKPILOT_ENABLE_ASAN "Instrument project code with AddressSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(CURL CONFIG REQUIRED)
find_package(unofficial-sqlite3 CONFIG REQUIRED)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(LibXml2 REQUIRED)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network)
add_library(maskpilot_options INTERFACE)
if(MSVC)
  target_compile_options(maskpilot_options INTERFACE /W4 /permissive- /utf-8 /EHsc)
  target_compile_options(maskpilot_options INTERFACE "$<$<CONFIG:Release>:/Zi>" /FS)
  target_link_options(maskpilot_options INTERFACE "$<$<CONFIG:Release>:/DEBUG:FULL>" "$<$<CONFIG:Release>:/OPT:REF>" "$<$<CONFIG:Release>:/OPT:ICF>")
  if(MASKPILOT_ENABLE_ASAN)
    target_compile_options(maskpilot_options INTERFACE /fsanitize=address)
    target_link_options(maskpilot_options INTERFACE /INCREMENTAL:NO)
  endif()
  target_compile_definitions(maskpilot_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
else()
  target_compile_options(maskpilot_options INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_subdirectory(src/domain)
add_subdirectory(src/application)
add_subdirectory(src/infrastructure)
foreach(tool ffmpeg ffprobe)
  add_executable(maskpilot_media_${tool} apps/media_guard/main.cpp)
  set_target_properties(maskpilot_media_${tool} PROPERTIES
    OUTPUT_NAME ${tool} RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/media_guard"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(maskpilot_media_${tool} PRIVATE maskpilot_options)
endforeach()
target_compile_definitions(maskpilot_media_ffprobe PRIVATE MASKPILOT_MEDIA_PROBE)
add_executable(maskpilot_cli apps/cli/main.cpp apps/cli/site_commands.cpp)
add_dependencies(maskpilot_cli maskpilot_media_ffmpeg maskpilot_media_ffprobe)
target_link_libraries(maskpilot_cli PRIVATE maskpilot_application maskpilot_infrastructure)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
add_subdirectory(src/ui)
add_executable(maskpilot WIN32 apps/desktop/main.cpp resources/app.qrc)
if(WIN32)
  target_sources(maskpilot PRIVATE resources/windows/app.manifest)
endif()
target_link_libraries(maskpilot PRIVATE maskpilot_ui maskpilot_infrastructure Qt6::Network)
add_dependencies(maskpilot maskpilot_media_ffmpeg maskpilot_media_ffprobe)
add_executable(maskpilot_native_host apps/native_host/main.cpp)
target_link_libraries(maskpilot_native_host PRIVATE Qt6::Core Qt6::Network maskpilot_options)
if(WIN32)
  enable_language(RC)
  foreach(target maskpilot maskpilot_cli maskpilot_native_host maskpilot_media_ffmpeg maskpilot_media_ffprobe)
    get_target_property(binary_name ${target} OUTPUT_NAME)
    if(NOT binary_name)
      set(binary_name ${target})
    endif()
    set(MASKPILOT_BINARY_NAME "${binary_name}.exe")
    configure_file(resources/windows/version.rc.in "${CMAKE_BINARY_DIR}/generated/${target}-version.rc" @ONLY)
    target_sources(${target} PRIVATE "${CMAKE_BINARY_DIR}/generated/${target}-version.rc")
  endforeach()
endif()
if(MSVC)
  foreach(target maskpilot maskpilot_cli maskpilot_native_host)
    target_link_options(${target} PRIVATE "/PDBALTPATH:$<TARGET_PDB_FILE_NAME:${target}>")
  endforeach()
endif()
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_package(GTest CONFIG REQUIRED)
  add_subdirectory(tests)
  target_sources(maskpilot_tests PRIVATE tests/unit/service_test.cpp)
  target_link_libraries(maskpilot_tests PRIVATE maskpilot_ui)
  target_sources(maskpilot_tests PRIVATE tests/unit/media_test.cpp)
  target_sources(maskpilot_tests PRIVATE tests/integration/torrent_test.cpp)
  target_link_libraries(maskpilot_tests PRIVATE LibtorrentRasterbar::torrent-rasterbar unofficial::sqlite3::sqlite3)
  add_dependencies(maskpilot_tests maskpilot_cli)
  target_compile_definitions(maskpilot_tests PRIVATE
    MASKPILOT_TEST_PYTHON="${Python3_EXECUTABLE}"
    MASKPILOT_TEST_PROXY_FIXTURE="${CMAKE_SOURCE_DIR}/tests/support/proxy_fixture.py")
  add_test(NAME native_bridge_product_path COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/native_host_test.py $<TARGET_FILE:maskpilot_native_host> $<TARGET_FILE:maskpilot>)
  set_tests_properties(native_bridge_product_path PROPERTIES TIMEOUT 90)
  add_test(NAME scoped_native_bridge_product_path COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/context_bridge_test.py $<TARGET_FILE:maskpilot_native_host> $<TARGET_FILE:maskpilot>)
  set_tests_properties(scoped_native_bridge_product_path PROPERTIES TIMEOUT 180)
  add_test(NAME stress_report_publication_failure COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/system/stress_report_test.py $<TARGET_FILE:maskpilot_native_host> $<TARGET_FILE:maskpilot>)
  set_tests_properties(stress_report_publication_failure PROPERTIES TIMEOUT 90)
  add_test(NAME ftp_product_path COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/support/ftp_fixture.py --exercise $<TARGET_FILE:maskpilot_cli>)
  set_tests_properties(ftp_product_path PROPERTIES TIMEOUT 300)
  add_test(NAME proxy_product_path COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration/proxy_test.py --exercise $<TARGET_FILE:maskpilot_cli>)
  set_tests_properties(proxy_product_path PROPERTIES TIMEOUT 300)
endif()
